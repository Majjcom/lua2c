file(GLOB LUA_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/lua/*.c")
list(APPEND LUA_SOURCES "${PROJECT_SOURCE_DIR}/runtime/lua2c_runtime.c")
if(LUA2C_SHARED_RUNTIME)
    add_library(lualib SHARED ${LUA_SOURCES})
    target_compile_definitions(lualib PUBLIC LUA2C_RUNTIME_SHARED)
    if(WIN32)
        target_compile_definitions(lualib PUBLIC LUA_BUILD_AS_DLL)
    endif()
else()
    add_library(lualib STATIC ${LUA_SOURCES})
endif()
add_library(lua2c::runtime ALIAS lualib)
set_target_properties(lualib PROPERTIES EXPORT_NAME runtime POSITION_INDEPENDENT_CODE ON)
target_include_directories(lualib PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/lua>
    $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/runtime>
    $<BUILD_INTERFACE:${PROJECT_BINARY_DIR}/generated>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/lua2c/runtime>)
if(LUA2C_RUNTIME_ASSERTIONS)
    target_compile_definitions(lualib PUBLIC LUAI_ASSERT LUA_USE_APICHECK)
endif()
if(UNIX)
    target_compile_definitions(lualib PRIVATE LUA_USE_POSIX LUA_USE_DLOPEN)
    target_link_libraries(lualib PUBLIC ${CMAKE_DL_LIBS} m)
endif()
if(MSVC)
    target_compile_definitions(lualib PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_executable(lua exec/lua.c)
target_link_libraries(lua PRIVATE lualib)
add_executable(luac exec/luac.c)
target_link_libraries(luac PRIVATE lualib)
install(TARGETS lua luac RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
